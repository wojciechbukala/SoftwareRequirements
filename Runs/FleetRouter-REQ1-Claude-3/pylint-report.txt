DEPARTURE_MIN = 8 * 60   # 08:00 in minutes from midnight
MAX_WORK_MIN = 8 * 60    # 480 minutes = 8 hours


def parse_time(s):
    """Parse HH:MM string to minutes from midnight."""
    parts = s.strip().split(':')
    if len(parts) != 2:
        raise ValueError(f"Invalid time format: {s}")
    return int(parts[0]) * 60 + int(parts[1])


def fmt_time(mins):
    """Format minutes from midnight to HH:MM string."""
    return f"{int(mins) // 60:02d}:{int(mins) % 60:02d}"


def read_csv(path):
    """Read a CSV file and return list of dicts."""
    with open(path, newline='', encoding='utf-8-sig') as f:
        return list(csv.DictReader(f))


def write_csv(path, fieldnames, rows):
    """Write rows to a CSV file."""
    with open(path, 'w', newline='', encoding='utf-8') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)


def simulate_route(depot, pkg_sequence, distances):
    """
    Simulate a route from depot through pkg_sequence and back to depot.

    Returns (result_dict, None) on success, or (None, reason_str) on failure.
    result_dict has: stops, total_distance, total_time_min
    Each stop has: package_id, location_id, arrival_min, departure_min
    """
    time = DEPARTURE_MIN
    loc = depot
    dist_total = 0.0
    stops = []

    for pkg in pkg_sequence:
        dest = pkg['destination_id']
        info = distances.get((loc, dest))
        if info is None:
            return None, 'UNREACHABLE'

        arrival = time + info['travel_time_min']

        if arrival > pkg['tw_close_min']:
            return None, 'TIME_WINDOW'

        departure = max(arrival, pkg['tw_open_min']) + pkg['service_min']

        stops.append({
            'package_id': pkg['package_id'],
            'location_id': dest,
            'arrival_min': arrival,
            'departure_min': departure,
        })

        dist_total += info['distance_km']
        time = departure
        loc = dest

    # Return to depot
    if loc != depot:
        info = distances.get((loc, depot))
        if info is None:
            return None, 'UNREACHABLE'
        dist_total += info['distance_km']
        total_time = time + info['travel_time_min'] - DEPARTURE_MIN
    else:
        total_time = time - DEPARTURE_MIN

    if total_time > MAX_WORK_MIN:
        return None, 'MAX_DRIVER_TIME'

    return {
        'stops': stops,
        'total_distance': dist_total,
        'total_time_min': int(total_time),
    }, None


def optimize_2opt(pkg_sequence, depot, distances):
    """Apply 2-opt improvement to reduce route distance."""
    if len(pkg_sequence) < 3:
        return pkg_sequence

    best = pkg_sequence[:]
    base, _ = simulate_route(depot, best, distances)
    if base is None:
        return pkg_sequence
    best_dist = base['total_distance']
    best_time = base['total_time_min']
    improved = True

    while improved:
        improved = False
        for i in range(len(best) - 1):
            for j in range(i + 2, len(best)):
                candidate = best[:i] + best[i:j + 1][::-1] + best[j + 1:]
                result, _ = simulate_route(depot, candidate, distances)
                if result is None:
                    continue
                if (result['total_distance'] < best_dist or
                        (result['total_distance'] == best_dist and
                         result['total_time_min'] < best_time)):
                    best_dist = result['total_distance']
                    best_time = result['total_time_min']
                    best = candidate
                    improved = True
    return best


def get_volume_field(row):
    """Get volume value handling both 'volume_m³' and 'volume_m3' field names."""
    for key in row:
        if key.startswith('volume_m'):
            return float(row[key])
    raise KeyError("No volume field found in row")


def main():
    parser = argparse.ArgumentParser(description='FleetRouter - Daily route planner')
    parser.add_argument('--input', required=True, help='Input directory')
    parser.add_argument('--output', required=True, help='Output directory')
    args = parser.parse_args()

    input_dir = args.input
    output_dir = args.output

    # ── Load locations ──────────────────────────────────────────────────────
    locations = {}
    for row in read_csv(os.path.join(input_dir, 'locations.csv')):
        locations[row['location_id']] = row['name']

    # ── Load distances ──────────────────────────────────────────────────────
    distances = {}
    for row in read_csv(os.path.join(input_dir, 'distances.csv')):
        key = (row['from_location_id'], row['to_location_id'])
        distances[key] = {
            'distance_km': float(row['distance_km']),
            'travel_time_min': int(row['travel_time_min']),
        }

    # ── Load vehicles, excluding those with unknown depots ──────────────────
    vehicles = []
    for row in read_csv(os.path.join(input_dir, 'vehicles.csv')):
        if row['depot_location_id'] not in locations:
            continue
        vehicles.append({
            'vehicle_id': row['vehicle_id'],
            'max_weight_kg': float(row['max_weight_kg']),
            'max_volume_m3': float(row['max_volume_m3']),
            'depot_location_id': row['depot_location_id'],
        })

    # ── Load packages ───────────────────────────────────────────────────────
    packages = []
    undeliverable = []

    for row in read_csv(os.path.join(input_dir, 'packages.csv')):
        pkg_id = row['package_id']
        dest_id = row['destination_id']

        # Validate destination location
        if dest_id not in locations:
            undeliverable.append({'package_id': pkg_id, 'reason': 'UNREACHABLE'})
            continue

        # Parse and validate time window
        try:
            tw_open = parse_time(row['tw_open'])
            tw_close = parse_time(row['tw_close'])
        except Exception:
            undeliverable.append({'package_id': pkg_id, 'reason': 'TIME_WINDOW'})
            continue

        if tw_open >= tw_close:
            undeliverable.append({'package_id': pkg_id, 'reason': 'TIME_WINDOW'})
            continue

        try:
            volume = get_volume_field(row)
        except (KeyError, ValueError):
            undeliverable.append({'package_id': pkg_id, 'reason': 'NO_VEHICLE'})
            continue

        packages.append({
            'package_id': pkg_id,
            'destination_id': dest_id,
            'weight_kg': float(row['weight_kg']),
            'volume_m3': volume,
            'tw_open': row['tw_open'],
            'tw_close': row['tw_close'],
            'tw_open_min': tw_open,
            'tw_close_min': tw_close,
            'service_min': int(row['service_min']),
            'priority': int(row['priority']),
        })

    # ── Check UNREACHABLE (missing distance entry from/to any depot) ─────────
    reachable_packages = []
    for pkg in packages:
        dest = pkg['destination_id']
        can_reach = any(
            (v['depot_location_id'], dest) in distances and
            (dest, v['depot_location_id']) in distances
            for v in vehicles
        )
        if not can_reach:
            undeliverable.append({'package_id': pkg['package_id'], 'reason': 'UNREACHABLE'})
        else:
            reachable_packages.append(pkg)
    packages = reachable_packages

    # ── Sort: priority=1 first, then by tw_open ──────────────────────────────
    packages.sort(key=lambda p: (-p['priority'], p['tw_open_min']))

    # ── Initialize vehicle route states ──────────────────────────────────────
    vehicle_routes = {
        v['vehicle_id']: {
            'vehicle': v,
            'depot': v['depot_location_id'],
            'packages': [],
            'total_weight': 0.0,
            'total_volume': 0.0,
        }
        for v in vehicles
    }

    # ── Greedy cheapest insertion ─────────────────────────────────────────────
    for pkg in packages:
        weight_ok_count = 0
        cap_ok_count = 0
        tw_fail = False
        time_fail = False

        best_vehicle_id = None
        best_pos = None
        best_cost = float('inf')
        best_route_time = float('inf')

        for vid, vstate in vehicle_routes.items():
            v = vstate['vehicle']

            # Weight capacity check
            if vstate['total_weight'] + pkg['weight_kg'] > v['max_weight_kg']:
                continue
            weight_ok_count += 1

            # Volume capacity check
            if vstate['total_volume'] + pkg['volume_m3'] > v['max_volume_m3']:
                continue
            cap_ok_count += 1

            # Compute base distance for this vehicle's current route
            base_result, _ = simulate_route(vstate['depot'], vstate['packages'], distances)
            base_dist = base_result['total_distance'] if base_result else 0.0

            # Try each insertion position
            for i in range(len(vstate['packages']) + 1):
                test_pkgs = vstate['packages'][:i] + [pkg] + vstate['packages'][i:]
                result, fail_reason = simulate_route(vstate['depot'], test_pkgs, distances)

                if result is not None:
                    added_dist = result['total_distance'] - base_dist
                    if (added_dist < best_cost or
                            (added_dist == best_cost and
                             result['total_time_min'] < best_route_time)):
                        best_cost = added_dist
                        best_route_time = result['total_time_min']
                        best_pos = i
                        best_vehicle_id = vid
                else:
                    if fail_reason == 'TIME_WINDOW':
                        tw_fail = True
                    elif fail_reason == 'MAX_DRIVER_TIME':
                        time_fail = True

        if best_vehicle_id is not None:
            vstate = vehicle_routes[best_vehicle_id]
            vstate['packages'].insert(best_pos, pkg)
            vstate['total_weight'] += pkg['weight_kg']
            vstate['total_volume'] += pkg['volume_m3']
        else:
            # Determine rejection reason
            if weight_ok_count == 0:
                reason = 'CAPACITY_WEIGHT'
            elif cap_ok_count == 0:
                reason = 'CAPACITY_VOLUME'
            elif tw_fail:
                reason = 'TIME_WINDOW'
            elif time_fail:
                reason = 'MAX_DRIVER_TIME'
            else:
                reason = 'NO_VEHICLE'
            undeliverable.append({'package_id': pkg['package_id'], 'reason': reason})

    # ── 2-opt route optimization ──────────────────────────────────────────────
    for vid, vstate in vehicle_routes.items():
        if len(vstate['packages']) > 2:
            vstate['packages'] = optimize_2opt(vstate['packages'], vstate['depot'], distances)

    # ── Generate output ───────────────────────────────────────────────────────
    os.makedirs(output_dir, exist_ok=True)

    stops_rows = []
    summary_rows = []

    for idx, v in enumerate(vehicles, start=1):
        vid = v['vehicle_id']
        vstate = vehicle_routes[vid]
        route_id = f"R{idx:03d}"
        depot = vstate['depot']
        pkgs = vstate['packages']

        if pkgs:
            result, _ = simulate_route(depot, pkgs, distances)
        else:
            result = {'stops': [], 'total_distance': 0.0, 'total_time_min': 0}

        total_dist = result['total_distance'] if result else 0.0
        total_time = result['total_time_min'] if result else 0

        for pos, stop in enumerate(result['stops'] if result else [], start=1):
            stops_rows.append({
                'route_id': route_id,
                'vehicle_id': vid,
                'stop_position_in_order': pos,
                'location_id': stop['location_id'],
                'delivered_id': stop['package_id'],
                'arrival_time': fmt_time(stop['arrival_min']),
                'departure_time': fmt_time(stop['departure_min']),
            })

        summary_rows.append({
            'vehicle_id': vid,
            'total_distance_km': round(total_dist, 2),
            'total_time_min': int(total_time),
            'packages_delivered': len(pkgs),
        })

    write_csv(
        os.path.join(output_dir, 'stops_order.csv'),
        ['route_id', 'vehicle_id', 'stop_position_in_order', 'location_id',
         'delivered_id', 'arrival_time', 'departure_time'],
        stops_rows,
    )

    write_csv(
        os.path.join(output_dir, 'undeliverable.csv'),
        ['package_id', 'reason'],
        undeliverable,
    )

    write_csv(
        os.path.join(output_dir, 'summary.csv'),
        ['vehicle_id', 'total_distance_km', 'total_time_min', 'packages_delivered'],
        summary_rows,
    )

    delivered = sum(len(v['packages']) for v in vehicle_routes.values())
    print(f"FleetRouter complete: {delivered} packages delivered, "
          f"{len(undeliverable)} undeliverable.")


if __name__ == '__main__':
    main()
